Migration 3 — Atomic Booking RPC
new 20260910170000_atomic_booking_rpc.sql FUNCTION SECURITY DEFINER 
-- Fixes: race condition where 3 separate queries could allow double-bookings.
-- This function runs entirely inside a single transaction.
CREATE OR REPLACE FUNCTION public.book_tee_time(
  p_tee_time_id uuid,
  p_user_id     uuid,
  p_players     int
)
RETURNS public.bookings
LANGUAGE plpgsql
SECURITY DEFINER          -- runs as owner, bypasses RLS on tee_times
SET search_path = public
AS $$
DECLARE
  v_booking public.bookings;
BEGIN
  -- Atomically check AND decrement in one statement.
  -- If another request already took the last spot, NOT FOUND fires.
  UPDATE public.tee_times
  SET
    available_spots = available_spots - p_players,
    status          = CASE WHEN (available_spots - p_players) = 0
                      THEN 'full' ELSE 'available' END
  WHERE id              = p_tee_time_id
    AND status         <> 'full'
    AND available_spots >= p_players;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Not enough spots available'
      USING ERRCODE = 'P0001';
  END IF;

  INSERT INTO public.bookings (tee_time_id, user_id, players)
  VALUES (p_tee_time_id, p_user_id, p_players)
  RETURNING * INTO v_booking;

  RETURN v_booking;
END;
$$;

REVOKE EXECUTE ON FUNCTION public.book_tee_time(uuid, uuid, int) FROM PUBLIC;
GRANT  EXECUTE ON FUNCTION public.book_tee_time(uuid, uuid, int) TO authenticated;

Migration 5 — Rolling Tee Time Refresh
new 20260910170002_rolling_tee_times.sql FUNCTION rolling refresh 
-- Fixes: seeded tee times were generated once at migration time with now(),
-- meaning they expired after 7 days and the app would show zero results.
-- Schedule SELECT public.refresh_tee_times() daily via pg_cron or Edge Function cron.
CREATE OR REPLACE FUNCTION public.refresh_tee_times()
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_course     record;
  v_slot_time  timestamptz;
  v_spots      int;
  v_slots      text[] := ARRAY['07:00','07:20',…,'14:00']; -- 24 slots/day
BEGIN
  -- 1. Purge: delete past slots and anything beyond 7 days out
  DELETE FROM public.tee_times
  WHERE start_time <  date_trunc('day', now())
     OR start_time >= date_trunc('day', now()) + '8 days'::interval;

  -- 2. Fill: insert slots for today + 7 days, skipping existing rows (idempotent)
  FOR v_course IN SELECT id, price_from FROM public.courses
                  WHERE data_source = 'live' LOOP
    FOR v_day_offset IN 0..6 LOOP
      FOREACH v_slot IN ARRAY v_slots LOOP
        v_slot_time := (CURRENT_DATE + v_day_offset || ' ' || v_slot)::timestamptz;

        IF NOT EXISTS (
          SELECT 1 FROM public.tee_times
          WHERE course_id = v_course.id AND start_time = v_slot_time
        ) THEN
          v_spots := GREATEST(0, 4 - floor(random() * 5)::int);
          INSERT INTO public.tee_times (course_id, start_time, price, available_spots, max_spots, status)
          VALUES (
            v_course.id, v_slot_time,
            v_course.price_from + floor(random() * 40)::int,
            v_spots, 4,
            CASE WHEN v_spots = 0 THEN 'full' ELSE 'available' END
          );
        END IF;
      END LOOP;
    END LOOP;
  END LOOP;
END;
$$;

REVOKE EXECUTE ON FUNCTION public.refresh_tee_times() FROM PUBLIC;
GRANT  EXECUTE ON FUNCTION public.refresh_tee_times() TO service_role;

SELECT public.refresh_tee_times(); -- run immediately after migration
